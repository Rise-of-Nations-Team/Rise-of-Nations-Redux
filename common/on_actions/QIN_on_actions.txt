on_actions = {
	on_startup = {

    # Event trigger related
		effect = {
			limit = {
				has_start_date < 1900.1.1
			}
			# Boxers take over Beijing
			every_country = {
				limit = {
					OR = {
						has_idea = DNC_eight_nation_alliance
						original_tag = QIN
					}
				}
				country_event = {
					id = dnc_boxers.3
					days = 156
				}
			}
			# Summer rains event
			every_country = {
				limit = {
					OR = {
						has_idea = DNC_eight_nation_alliance
						original_tag = QIN
					}
				}
				country_event = {
					id = dnc_boxers.4
					days = 212
					random_days = 30
				}
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					tag = QIN
					FROM = {
						every_country = {
							limit = {
								has_ideas = DNC_eight_nation_alliance
							}
						}
					}
					FROM.FROM = {
						AND = {
							state = 1164
							#state = Tientsin
						}
					}
				}
				FROM = {
					white_peace = QIN
					every_state = {
						limit = {
							is_core_of = QIN
							NOT = {
								is_owned_by = QIN
							}
						}
						transfer_state_to = QIN
						add_global_flag = QIN_won_against_ENA_gf
					}
				}
			}
		}
	}
}
