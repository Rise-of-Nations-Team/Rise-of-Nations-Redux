### Colours a nationality's support number by the same thresholds the register itself
### reads every month (see AUH_nat_thresholds_<NAT> in RON_AUH_dnc_nationalities.txt):
### green at 35 and above, yellow below that, red below 20.

defined_text = {
	name = GetAUS_supportColor
	text = { trigger = { check_variable = { var = AUS_support value = 20 compare = less_than } } localization_key = R }
	text = { trigger = { check_variable = { var = AUS_support value = 35 compare = less_than } } localization_key = Y }
	text = { localization_key = G }
}
