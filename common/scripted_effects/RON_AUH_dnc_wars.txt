AUH_war_conduct_common = {
	hidden_effect = {
		set_variable = { AUH_war_conduct = 0 }
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_arm_the_new_military_law_passed
			}
		}
		custom_effect_tooltip = AUH_kf_conduct_no_army_bill_tt
		hidden_effect = {
			add_to_variable = { var = AUH_war_conduct value = 1 }
		}
	}

	if = {
		limit = {
			OR = {
				has_idea = AUH_the_frozen_contingent_1
				has_idea = AUH_the_frozen_contingent_2
				has_idea = AUH_the_frozen_contingent_3
				NOT = {
					has_completed_focus = AUHDNC_arm_the_recruit_contingent
				}
			}
		}
		custom_effect_tooltip = AUH_kf_conduct_frozen_contingent_tt
		hidden_effect = {
			add_to_variable = { var = AUH_war_conduct value = 1 }
		}
	}
}

AUH_redl_poison = {
	if = {
		limit = { has_country_flag = AUH_redl_unmasked }
		subtract_from_variable = { var = AUH_war_conduct value = 1 }
		country_event = { id = auh_clock.319 days = 4 }
	}

	else_if = {
		limit = {
			check_variable = {
				var = AUH_redl_ledger
				value = 16
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { var = AUH_war_conduct value = 2 }
	}

	else_if = {
		limit = {
			check_variable = {
				var = AUH_redl_ledger
				value = 8
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { var = AUH_war_conduct value = 1 }
	}
}

AUH_war_opening_route = {
	hidden_effect = {
		set_country_flag = AUH_kf_war_run
		clamp_variable = {
			var = AUH_war_conduct
			min = 0
			max = 4
		}
		if = {
			limit = {
				check_variable = {
					var = AUH_war_conduct
					value = 0
					compare = equals
				}
			}
			country_event = { id = auh_clock.315 days = 9 }
		}
		else_if = {
			limit = {
				check_variable = {
					var = AUH_war_conduct
					value = 1
					compare = equals
				}
			}
			country_event = { id = auh_clock.316 days = 9 }
		}
		else_if = {
			limit = {
				check_variable = {
					var = AUH_war_conduct
					value = 2
					compare = equals
				}
			}
			country_event = { id = auh_clock.317 days = 9 }
		}
		else = {
			country_event = { id = auh_clock.318 days = 9 }
		}
	}
}

AUH_war_clock_check = {
	if = {
		limit = { has_country_flag = AUH_kf_corridor_held }
		add_to_variable = { var = AUH_kf_corridor_drain value = -0.005 }
		add_to_variable = {
			var = AUH_kf_corridor_manpower
			value = -0.005
		}
		clamp_variable = {
			var = AUH_kf_corridor_drain
			min = -0.30
			max = 0
		}
		clamp_variable = {
			var = AUH_kf_corridor_manpower
			min = -0.30
			max = 0
		}
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_kf_peace_italy_offered
			}
			country_exists = ITA
			has_war_with = ITA
			ITA = {
				OR = {
					has_capitulated = yes
					surrender_progress > 0.4
				}
			}
		}
		set_country_flag = AUH_kf_peace_italy_offered
		country_event = { id = auh_clock.292 days = 3 }
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_kf_peace_corridor_offered
			}
			country_exists = OTT
			has_war_with = OTT
			OR = {
				controls_state = 731
				OTT = {
					OR = {
						has_capitulated = yes
						surrender_progress > 0.35
					}
				}
			}
		}
		set_country_flag = AUH_kf_peace_corridor_offered
		country_event = { id = auh_clock.302 days = 3 }
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_kf_peace_russia_offered
			}
			country_exists = RUS
			has_war_with = RUS
			RUS = {
				OR = {
					has_capitulated = yes
					surrender_progress > 0.5
				}
			}
		}
		set_country_flag = AUH_kf_peace_russia_offered
		country_event = { id = auh_clock.308 days = 3 }
	}

	if = {
		limit = {
			NOT = { has_country_flag = AUH_kf_east_lost }
			country_exists = RUS
			has_war_with = RUS
			surrender_progress > 0.4
		}
		set_country_flag = AUH_kf_east_lost
		country_event = { id = auh_clock.309 days = 3 }
	}

	if = {
		limit = {
			NOT = { has_country_flag = AUH_kf_league_formed }
			has_country_flag = AUH_kf_corridor_held
			country_exists = SER
			check_variable = {
				var = AUH_clock_months
				value = 96
				compare = greater_than_or_equals
			}
		}
		random_list = {
			80 = { }
			20 = {
				set_country_flag = AUH_kf_league_formed
				country_event = { id = auh_clock.300 days = 5 }
			}
		}
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_kf_petersburg_answered
			}
			check_variable = {
				var = AUH_russia_anger
				value = 8
				compare = greater_than_or_equals
			}
		}
		set_country_flag = AUH_kf_petersburg_answered
		country_event = { id = auh_clock.299 days = 6 }
	}

	if = {
		limit = {
			NOT = { has_country_flag = AUH_redl_unmasked }
			check_variable = {
				var = AUH_redl_ledger
				value = 10
				compare = greater_than_or_equals
			}
			check_variable = {
				var = AUH_redl_smoke
				value = 2
				compare = less_than
			}
		}
		random_list = {
			92 = { }
			8 = {
				country_event = { id = auh_clock.287 days = 4 }
			}
		}
	}
}
