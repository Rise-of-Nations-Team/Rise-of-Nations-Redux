QIN_box_war_revolutionaries_setup_se = {
	if = {
		limit = {
			check_variable = { QIN_box_war_chosen_revolution = 1 }
			NOT = {
				has_global_flag = QIN_mongol_revolution
			}
		}
		all_states = {
			limit = {
				is_owned_by = QIN
				is_core_of = MON
			}
			transfer_state_to = MON
		}
		329 = {
			add_core_of = MON

			set_capital = {
