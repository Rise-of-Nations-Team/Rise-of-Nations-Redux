AUH_z_strain_months_tick = {
	if = {
		limit = {
			check_variable = {
				var = AUH_treasury_strain
				value = 70
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { var = AUH_z_strain_months value = 1 }
	}

	else = {
		set_variable = { AUH_z_strain_months = 0 }
	}
	clamp_variable = {
		var = AUH_z_strain_months
		min = 0
		max = 60
	}
}

AUH_z_stage_one_check = {
	if = {
		limit = {
			NOT = { has_country_flag = AUH_dnc_collapse_running }
			NOT = { has_country_flag = AUH_dnc_era_end }
			OR = {
				check_variable = {
					var = AUH_nat_rising_count
					value = 3
					compare = greater_than_or_equals
				}
				check_variable = {
					var = AUH_z_strain_months
					value = 6
					compare = greater_than_or_equals
				}
				check_variable = {
					var = AUH_treasury_strain
					value = 80
					compare = greater_than_or_equals
				}
				check_variable = {
					var = AUH_nat_secessions
					value = 2
					compare = greater_than_or_equals
				}
				has_capitulated = yes
				surrender_progress > 0.35
				has_country_flag = AUH_kf_east_lost
			}
		}
		AUH_z_collapse_open = yes
	}
}
