########################################################################################
# AUSTRIA-HUNGARY - THE AUSGLEICH ENGINE
########################################################################################
# AUH_gemeinsamkeit 0-100   what the two halves still hold in common
# AUH_quote 25-70           the kingdom's share of the common charge
# AUH_months_to_renewal     the clock, counted in months, never in years
# AUH_ausgleich_months      the same clock under the name the focus tree reads
# AUH_ausgleich_tier 1-5    derived from the Community, never set by hand

# Clamps everything this engine writes to
AUH_dnc_ausgleich_clamp = {
	clamp_variable = { var = AUH_gemeinsamkeit min = 0 max = 100 }
	clamp_variable = { var = AUH_quote min = 25 max = 70 }
	clamp_variable = { var = HUN_support min = 0 max = 100 }
	clamp_variable = { var = AUS_support min = 0 max = 100 }
	clamp_variable = { var = CZE_support min = 0 max = 100 }
	clamp_variable = { var = CRO_support min = 0 max = 100 }
}

# Wears one settlement spirit and only one
AUH_dnc_wear_the_settlement = {
	if = {
		limit = { has_idea = AUH_dnc_the_personal_union }
		remove_ideas = AUH_dnc_the_personal_union
	}
	if = {
		limit = { has_idea = AUH_dnc_the_hungarian_bargain }
		remove_ideas = AUH_dnc_the_hungarian_bargain
	}
	if = {
		limit = { has_idea = AUH_dnc_the_two_parliaments }
		remove_ideas = AUH_dnc_the_two_parliaments
	}
	if = {
		limit = { has_idea = AUH_dnc_the_indivisible_army }
		remove_ideas = AUH_dnc_the_indivisible_army
	}
	if = {
		limit = { has_idea = AUH_dnc_the_einheitsreich }
		remove_ideas = AUH_dnc_the_einheitsreich
	}
	add_ideas = $SPIRIT$
}

# Reads the Community off and puts on the spirit it calls for
AUH_dnc_refresh_the_settlement = {
	AUH_dnc_ausgleich_clamp = yes

	set_temp_variable = { AUH_tier_wanted = 1 }
	if = {
		limit = { check_variable = { var = AUH_gemeinsamkeit value = 20 compare = greater_than_or_equals } }
		set_temp_variable = { AUH_tier_wanted = 2 }
	}
	if = {
		limit = { check_variable = { var = AUH_gemeinsamkeit value = 40 compare = greater_than_or_equals } }
		set_temp_variable = { AUH_tier_wanted = 3 }
	}
	if = {
		limit = { check_variable = { var = AUH_gemeinsamkeit value = 60 compare = greater_than_or_equals } }
		set_temp_variable = { AUH_tier_wanted = 4 }
	}
	if = {
		limit = { check_variable = { var = AUH_gemeinsamkeit value = 80 compare = greater_than_or_equals } }
		set_temp_variable = { AUH_tier_wanted = 5 }
	}

	if = {
		limit = { NOT = { check_variable = { var = AUH_ausgleich_tier value = AUH_tier_wanted compare = equals } } }
		set_variable = { AUH_ausgleich_tier = AUH_tier_wanted }

		if = {
			limit = { check_variable = { var = AUH_ausgleich_tier value = 5 compare = equals } }
			AUH_dnc_wear_the_settlement = { SPIRIT = AUH_dnc_the_einheitsreich }
		}
		else_if = {
			limit = { check_variable = { var = AUH_ausgleich_tier value = 4 compare = equals } }
			AUH_dnc_wear_the_settlement = { SPIRIT = AUH_dnc_the_indivisible_army }
		}
		else_if = {
			limit = { check_variable = { var = AUH_ausgleich_tier value = 3 compare = equals } }
			AUH_dnc_wear_the_settlement = { SPIRIT = AUH_dnc_the_two_parliaments }
		}
		else_if = {
			limit = { check_variable = { var = AUH_ausgleich_tier value = 2 compare = equals } }
			AUH_dnc_wear_the_settlement = { SPIRIT = AUH_dnc_the_hungarian_bargain }
		}
		else = {
			AUH_dnc_wear_the_settlement = { SPIRIT = AUH_dnc_the_personal_union }
		}
	}
}

########################################################################################
# THE BILL
########################################################################################
# What carrying the settlement costs, and how it is likely to go

AUH_dnc_price_the_settlement = {
	set_variable = { AUH_renewal_cost = AUH_quote }
	multiply_variable = { AUH_renewal_cost = 3 }
	round_variable = AUH_renewal_cost
	clamp_variable = { var = AUH_renewal_cost min = 60 max = 150 }

	set_temp_variable = { AUH_odds = AUH_gemeinsamkeit }
	add_to_temp_variable = { AUH_odds = 25 }
	if = {
		limit = { has_stability > 0.6 }
		add_to_temp_variable = { AUH_odds = 15 }
	}
	if = {
		limit = { NOT = { has_stability > 0.4 } }
		subtract_from_temp_variable = { AUH_odds = 20 }
	}
	if = {
		limit = { check_variable = { var = HUN_support value = 60 compare = greater_than_or_equals } }
		add_to_temp_variable = { AUH_odds = 15 }
	}
	if = {
		limit = { check_variable = { var = HUN_support value = 35 compare = less_than } }
		subtract_from_temp_variable = { AUH_odds = 25 }
	}
	if = {
		limit = { has_idea = AUH_dnc_the_provisorium }
		subtract_from_temp_variable = { AUH_odds = 15 }
	}
	if = {
		limit = { has_idea = AUH_dnc_hungarian_national_resistance }
		subtract_from_temp_variable = { AUH_odds = 20 }
	}
	clamp_temp_variable = { var = AUH_odds min = 5 max = 95 }
	set_variable = { AUH_renewal_odds = AUH_odds }
	round_variable = AUH_renewal_odds
}

########################################################################################
# THE MONTHLY TICK
########################################################################################
# Nothing here reads the calendar

AUH_dnc_ausgleich_monthly = {
	# the tree may put a new term on the clock under its own name
	if = {
		limit = { NOT = { check_variable = { var = AUH_ausgleich_months value = AUH_ausgleich_clock_seen compare = equals } } }
		set_variable = { AUH_months_to_renewal = AUH_ausgleich_months }
		# a parked clock is a term nobody has negotiated yet, not a term that has expired
		if = {
			limit = { check_variable = { var = AUH_months_to_renewal value = 1 compare = less_than } }
			set_variable = { AUH_months_to_renewal = 48 }
		}
		set_variable = { AUH_renewal_window = 0 }
		set_variable = { AUH_renewal_armed = 1 }
	}

	if = {
		limit = { check_variable = { var = AUH_months_to_renewal value = 1 compare = greater_than_or_equals } }
		subtract_from_variable = { AUH_months_to_renewal = 1 }
	}
	clamp_variable = { var = AUH_months_to_renewal min = 0 max = 240 }

	set_variable = { AUH_ausgleich_months = AUH_months_to_renewal }
	set_variable = { AUH_ausgleich_clock_seen = AUH_months_to_renewal }

	# a settlement that never came eats the common stock
	if = {
		limit = { has_idea = AUH_dnc_the_provisorium }
		subtract_from_variable = { AUH_gemeinsamkeit = 0.1 }
	}
	if = {
		limit = { has_idea = AUH_dnc_two_customs_territories }
		subtract_from_variable = { AUH_gemeinsamkeit = 0.1 }
	}
	# the highest settlement holds nobody's affection
	if = {
		limit = { check_variable = { var = AUH_ausgleich_tier value = 5 compare = equals } }
		subtract_from_variable = { HUN_support = 1 }
		subtract_from_variable = { CRO_support = 1 }
	}

	AUH_dnc_price_the_settlement = yes
	AUH_dnc_refresh_the_settlement = yes
	AUH_dnc_ausgleich_watch = yes
	AUH_dnc_ausgleich_clamp = yes
}

########################################################################################
# THE WATCH
########################################################################################
# The term falling due, the counties striking, the union itself in question

AUH_dnc_ausgleich_watch = {
	if = {
		limit = {
			check_variable = { var = AUH_renewal_armed value = 1 compare = greater_than_or_equals }
			check_variable = { var = AUH_months_to_renewal value = 1 compare = less_than }
			check_variable = { var = AUH_renewal_window value = 1 compare = less_than }
			NOT = { has_country_flag = AUH_dnc_union_dissolved }
		}
		set_variable = { AUH_renewal_window = 1 }
		set_variable = { AUH_renewal_armed = 0 }
		country_event = { id = auh_ausgleich.1 hours = 6 }
	}

	if = {
		limit = { check_variable = { var = AUH_gemeinsamkeit value = 90 compare = greater_than_or_equals } }
		add_to_variable = { AUH_ceiling_months = 1 }
	}
	else = {
		set_variable = { AUH_ceiling_months = 0 }
	}
	if = {
		limit = {
			check_variable = { var = AUH_ceiling_months value = 12 compare = greater_than_or_equals }
			NOT = { has_idea = AUH_dnc_hungarian_national_resistance }
			NOT = { has_country_flag = AUH_dnc_union_dissolved }
		}
		set_variable = { AUH_ceiling_months = 0 }
		country_event = { id = auh_ausgleich.5 hours = 6 }
	}

	if = {
		limit = {
			NOT = { has_country_flag = AUH_dnc_floor_crisis }
			NOT = { has_country_flag = AUH_dnc_union_dissolved }
			check_variable = { var = AUH_missed_renewals value = 2 compare = greater_than_or_equals }
		}
		set_country_flag = AUH_dnc_floor_crisis
		country_event = { id = auh_ausgleich.4 hours = 6 }
	}
	if = {
		limit = {
			has_country_flag = AUH_dnc_floor_crisis
			check_variable = { var = AUH_missed_renewals value = 1 compare = less_than }
		}
		clr_country_flag = AUH_dnc_floor_crisis
	}
}

########################################################################################
# THE TWO ENDINGS OF A RENEWAL
########################################################################################
# AUH_next_term carries the length the chosen option bought

AUH_dnc_the_settlement_carried = {
	set_variable = { AUH_months_to_renewal = AUH_next_term }
	set_variable = { AUH_ausgleich_months = AUH_next_term }
	set_variable = { AUH_ausgleich_clock_seen = AUH_next_term }
	set_variable = { AUH_renewal_window = 0 }
	set_variable = { AUH_renewal_armed = 1 }
	set_variable = { AUH_missed_renewals = 0 }
	if = {
		limit = { has_idea = AUH_dnc_the_provisorium }
		remove_ideas = AUH_dnc_the_provisorium
	}
	AUH_dnc_price_the_settlement = yes
	AUH_dnc_refresh_the_settlement = yes
}

AUH_dnc_the_settlement_lapses = {
	set_variable = { AUH_months_to_renewal = AUH_next_term }
	set_variable = { AUH_ausgleich_months = AUH_next_term }
	set_variable = { AUH_ausgleich_clock_seen = AUH_next_term }
	set_variable = { AUH_renewal_window = 0 }
	set_variable = { AUH_renewal_armed = 1 }
	add_to_variable = { AUH_missed_renewals = 1 }
	AUH_dnc_price_the_settlement = yes
	AUH_dnc_refresh_the_settlement = yes
}
