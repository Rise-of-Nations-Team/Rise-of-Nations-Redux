# Rebellion: the states gathered in the rebel_state array rise against the administration
create_rebellion = {
	clear_array = RON_rebel_states
	for_each_loop = {
		array = rebel_state
		value = rebel_state_id
		add_to_array = { array = RON_rebel_states value = rebel_state_id }
	}
	if = {
		limit = { has_array_size = { array = RON_rebel_states size > 0 } }
		add_stability = -0.03
		add_war_support = -0.02
		add_manpower = -2000
		add_timed_idea = { idea = RON_a_rising_in_the_colonies days = 365 }
		every_owned_state = {
			limit = { is_in_array = { array = ROOT.RON_rebel_states value = THIS.id } }
			add_dynamic_modifier = { modifier = RON_state_in_revolt days = 365 }
		}
	}
	clear_array = RON_rebel_states
	clear_temp_array = rebel_state
}
