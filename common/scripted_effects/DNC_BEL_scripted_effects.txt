BEL_estate_month = {
	set_temp_variable = { BEL_t_scandal_drift = 0 }
	if = {
		limit = {
			check_variable = {
				var = BEL_congo_regime
				value = 3
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { BEL_t_scandal_drift = 0.6 }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_congo_regime
				value = 2
				compare = equals
			}
		}
		set_temp_variable = { BEL_t_scandal_drift = 0.4 }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_congo_regime
				value = 1
				compare = equals
			}
		}
		set_temp_variable = { BEL_t_scandal_drift = 0.1 }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_congo_regime
				value = 0
				compare = less_than_or_equals
			}
		}
		set_temp_variable = { BEL_t_scandal_drift = -0.5 }
	}

	set_temp_variable = { BEL_t_denial_push = BEL_denials }
	multiply_temp_variable = { BEL_t_denial_push = 0.15 }
	add_to_temp_variable = { BEL_t_scandal_drift = BEL_t_denial_push }
	if = {
		limit = { has_country_flag = BEL_commission_sits }
		add_to_temp_variable = { BEL_t_scandal_drift = 0.3 }
	}

	if = {
		limit = { has_country_flag = BEL_congo_annexed }
		subtract_from_temp_variable = { BEL_t_scandal_drift = 0.2 }
	}

	add_to_variable = { var = BEL_scandal value = BEL_t_scandal_drift }
	set_temp_variable = { BEL_t_chamber_drift = 0.1 }
	if = {
		limit = {
			check_variable = {
				var = BEL_scandal
				value = 40
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { BEL_t_chamber_drift = 0.25 }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_scandal
				value = 65
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { BEL_t_chamber_drift = 0.25 }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_scandal
				value = 85
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { BEL_t_chamber_drift = 0.25 }
	}

	set_temp_variable = { BEL_t_denial_whip = BEL_denials }
	multiply_temp_variable = { BEL_t_denial_whip = 0.1 }
	subtract_from_temp_variable = { BEL_t_chamber_drift = BEL_t_denial_whip }
	if = {
		limit = { has_country_flag = BEL_congo_refused }
		subtract_from_temp_variable = { BEL_t_chamber_drift = 0.15 }
	}

	if = {
		limit = { NOT = { has_country_flag = BEL_congo_annexed } }
		add_to_variable = { var = BEL_chamber_congo value = BEL_t_chamber_drift }
	}

	if = {
		limit = {
			check_variable = {
				var = BEL_vote_cooldown
				value = 0
				compare = greater_than
			}
		}
		subtract_from_variable = { var = BEL_vote_cooldown value = 1 }
	}
}
