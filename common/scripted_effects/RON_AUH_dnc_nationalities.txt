		max = 20

		max = 3

}

	add_to_variable = { var = AUH_coercion_clean_months value = 1 }
	if = {
		limit = {
			check_variable = {
				var = AUH_coercion_clean_months
				value = 12
				compare = greater_than_or_equals
			}
		}
		set_variable = { AUH_coercion_clean_months = 0 }
		subtract_from_variable = { var = AUH_coercion_ledger value = 1 }
		clamp_variable = {
			var = AUH_coercion_ledger
			min = 0
			max = 40
		}
	}
}

AUH_nat_state_clear = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_restless }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_restless }
	}

	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
	}

	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_in_revolt }
	}
}

AUH_nat_drift_AUS = {
	if = {
		limit = {
			check_variable = {
				var = AUS_support
				value = AUH_nat_band_erosion
				compare = less_than
			}
		}
		subtract_from_variable = { AUS_support = AUH_register_erosion }
		custom_effect_tooltip = AUH_nat_AUS_down_tt
	}

	else_if = {
		limit = {
			check_variable = {
				var = AUS_support
				value = AUH_nat_band_recovery
				compare = less_than
			}
			has_stability > 0.7
		}
		add_to_variable = { var = AUS_support value = 1 }
		custom_effect_tooltip = AUH_nat_AUS_up_tt
	}

	clamp_variable = {
		var = AUS_support
		min = 0
		max = 100
	}
}

				limit = {

			}
		}
		else_if = {
			limit = {
				owner = {
					check_variable = {

						compare = less_than
					}
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_in_revolt }

			}
			if = {
				limit = {
					has_dynamic_modifier = { modifier = AUH_nat_restless }
				}
				remove_dynamic_modifier = { modifier = AUH_nat_restless }
			}
			if = {
				limit = {
					NOT = {

					}
				}
				add_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }

		}
		else_if = {

		limit = {
			check_variable = {
				var = POL_support
				value = AUH_nat_band_erosion
				compare = less_than
			}
		}
		subtract_from_variable = { POL_support = AUH_register_erosion }
		custom_effect_tooltip = AUH_nat_POL_down_tt
	}

	else_if = {
		limit = {
			check_variable = {
				var = POL_support
				value = AUH_nat_band_recovery
				compare = less_than
			}
			has_stability > 0.7
		}
		add_to_variable = { var = POL_support value = 1 }
		custom_effect_tooltip = AUH_nat_POL_up_tt

		var = POL_support
		min = 0
		max = 100
	}
}

AUH_nat_thresholds_POL = {
	every_owned_state = {
		limit = { is_core_of = POL }
		if = {
			limit = {
				owner = {
					check_variable = {

			}

				limit = {

				remove_dynamic_modifier = { modifier = AUH_nat_restless }

				}
				remove_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
					}
				}
				add_dynamic_modifier = { modifier = AUH_nat_in_revolt }
			}
		}
		else_if = {
			limit = {
				owner = {
					check_variable = {
						var = POL_support
						value = AUH_nat_band_levy
						compare = less_than
					}
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
				}
			}
			if = {
				limit = {
					has_dynamic_modifier = { modifier = AUH_nat_restless }
				}
				remove_dynamic_modifier = { modifier = AUH_nat_restless }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
					}
				}
				add_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
			}
		}
		else_if = {
			limit = {
				owner = {
					check_variable = {
						var = POL_support
						value = AUH_nat_band_restless
						compare = less_than
					}
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
