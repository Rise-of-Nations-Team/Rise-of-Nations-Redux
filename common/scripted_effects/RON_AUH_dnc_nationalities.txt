AUH_nat_clear_all = {
	every_owned_state = { AUH_nat_state_clear = yes }
}

AUH_nat_state_clear = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_restless }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_restless }
	}

	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
	}

	if = {
		limit = {
			has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
		}
		remove_dynamic_modifier = { modifier = AUH_nat_in_revolt }
	}
}

AUH_nat_drift_AUS = {
	if = {
		limit = {
			check_variable = {
				var = AUS_support
				value = 35
				compare = less_than
			}
		}
		subtract_from_variable = { var = AUS_support value = 1 }
		custom_effect_tooltip = AUH_nat_AUS_down_tt
	}

	else_if = {
		limit = {
			check_variable = {
				var = AUS_support
				value = 50
				compare = less_than
			}
			has_stability > 0.7
		}
		add_to_variable = { var = AUS_support value = 1 }
		custom_effect_tooltip = AUH_nat_AUS_up_tt
	}

	clamp_variable = {
		var = AUS_support
		min = 0
		max = 100
	}
}

AUH_nat_thresholds_AUS = {
	every_owned_state = {
		limit = { is_core_of = AUS }
		if = {
			limit = {
				owner = {
					check_variable = {
						var = AUS_support
						value = 5
						compare = less_than
					}
				}
			}
			if = {
				limit = {
					has_dynamic_modifier = { modifier = AUH_nat_restless }
				}
				remove_dynamic_modifier = { modifier = AUH_nat_restless }
			}
			if = {
				limit = {
					has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
				}
				remove_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
					}
				}
				add_dynamic_modifier = { modifier = AUH_nat_in_revolt }
			}
		}
		else_if = {
			limit = {
				owner = {
					check_variable = {
						var = AUS_support
						value = 20
						compare = less_than
					}
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
				}
			}
			if = {
				limit = {
					has_dynamic_modifier = { modifier = AUH_nat_restless }
				}
				remove_dynamic_modifier = { modifier = AUH_nat_restless }
			}
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
					}
				}
				add_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
			}
		}
		else_if = {
			limit = {
				owner = {
					check_variable = {
						var = AUS_support
						value = 35
						compare = less_than
					}
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_in_revolt }
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_refuses_the_levy }
				}
				NOT = {
					has_dynamic_modifier = { modifier = AUH_nat_restless }
				}
			}
			add_dynamic_modifier = { modifier = AUH_nat_restless }
		}
	}
}
