AUH_house_prestige_dial = {
	set_variable = {
		AUH_house_prestige_stability = AUH_house_prestige
	}
	subtract_from_variable = { AUH_house_prestige_stability = 50 }
	divide_variable = { AUH_house_prestige_stability = 500 }
	set_variable = { AUH_house_prestige_pp = AUH_house_prestige }
	subtract_from_variable = { AUH_house_prestige_pp = 50 }
	divide_variable = { AUH_house_prestige_pp = 250 }
	set_variable = {
		AUH_house_prestige_drift = AUH_house_prestige
	}
	subtract_from_variable = { AUH_house_prestige_drift = 50 }
	divide_variable = { AUH_house_prestige_drift = 400 }
}

AUH_house_prestige_gain = {
	add_to_variable = { var = AUH_house_prestige value = 2 }
	clamp_variable = {
		var = AUH_house_prestige
		min = 0
		max = 100
	}
	AUH_house_prestige_dial = yes
}

AUH_house_prestige_loss = {
	subtract_from_variable = { var = AUH_house_prestige value = 4 }
	clamp_variable = {
		var = AUH_house_prestige
		min = 0
		max = 100
	}
	AUH_house_prestige_dial = yes
}

AUH_house_scandal_roll = {
	if = {
		limit = {
			NOT = { has_country_flag = AUH_c_scandal_pending }
			NOT = { has_country_flag = AUH_dnc_era_end }
			check_variable = {
				var = AUH_clock_months
				value = 4
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { AUH_t_house_chance = 5 }
		if = {
			limit = { has_country_flag = AUH_c_blind_eye }
			add_to_temp_variable = { AUH_t_house_chance = 4 }
		}
		if = {
			limit = { has_country_flag = AUH_c_house_law_teeth }
			subtract_from_temp_variable = { AUH_t_house_chance = 2 }
		}
		set_temp_variable = { AUH_t_house_grievance = AUH_heir_grievance }
		divide_temp_variable = { AUH_t_house_grievance = 2 }
		add_to_temp_variable = { AUH_t_house_chance = AUH_t_house_grievance }
		clamp_temp_variable = {
			var = AUH_t_house_chance
			min = 1
			max = 25
		}
		set_temp_variable = { AUH_t_house_roll = random }
		multiply_temp_variable = { AUH_t_house_roll = 100 }
		if = {
			limit = {
				check_variable = {
					var = AUH_t_house_roll
					value = AUH_t_house_chance
					compare = less_than
				}
			}
			set_country_flag = {
				flag = AUH_c_scandal_pending
				days = 150
				value = 1
			}
			country_event = { id = auh_clock.240 days = 3 }
		}
	}
}

AUH_house_otto_window = {
	if = {
		limit = {
			NOT = { has_country_flag = AUH_c_otto_is_dead }
			check_variable = {
				var = AUH_clock_months
				value = 78
				compare = greater_than_or_equals
			}
		}
		if = {
			limit = {
				check_variable = {
					var = AUH_clock_months
					value = 84
					compare = greater_than_or_equals
				}
			}
			country_event = { id = auh_clock.248 days = 2 }
		}
		else = {
			set_temp_variable = { AUH_t_otto_chance = 25 }
			set_temp_variable = { AUH_t_otto_hush = AUH_scandal_ledger }
			multiply_temp_variable = { AUH_t_otto_hush = 6 }
			add_to_temp_variable = { AUH_t_otto_chance = AUH_t_otto_hush }
			clamp_temp_variable = {
				var = AUH_t_otto_chance
				min = 10
				max = 80
			}
			set_temp_variable = { AUH_t_otto_roll = random }
			multiply_temp_variable = { AUH_t_otto_roll = 100 }
			if = {
				limit = {
					check_variable = {
						var = AUH_t_otto_roll
						value = AUH_t_otto_chance
						compare = less_than
					}
				}
				country_event = { id = auh_clock.248 days = 2 }
			}
		}
	}
}

AUH_house_pulse = {
	if = {
		limit = { has_country_flag = AUH_c_scandals_open }
		if = {
			limit = { has_country_flag = AUH_c_house_law_teeth }
			add_to_variable = { var = AUH_house_prestige value = 1 }
		}
		if = {
			limit = { has_country_flag = AUH_c_blind_eye }
			subtract_from_variable = { var = AUH_house_prestige value = 0.5 }
		}
		clamp_variable = {
			var = AUH_house_prestige
			min = 0
			max = 100
		}
		AUH_house_scandal_roll = yes
		AUH_house_otto_window = yes
	}

	if = {
		limit = { has_variable = AUH_house_prestige }
		AUH_house_prestige_dial = yes
	}
}

AUH_svc_mortar_progress_check = {
	clamp_variable = {
		var = AUH_mortar_progress
		min = 0
		max = 100
	}

	if = {
		limit = {
			NOT = {
				has_country_flag = AUH_arm_the_mortar_delivered
			}
			check_variable = {
				var = AUH_mortar_progress
				value = 100
				compare = greater_than_or_equals
			}
		}
		set_country_flag = AUH_arm_the_mortar_delivered
		country_event = { id = auh_clock.226 days = 5 }
	}
}
