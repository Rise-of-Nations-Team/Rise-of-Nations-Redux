############################################
# Operational Pause — hold all fronts during pause window
# Priority 2000: overrides posture but yields to crisis and wardec holds.
# Ratio follows EAI's working front hold pattern; ratio 0.0 can fail to bind to real fronts.
############################################
FAI_operational_pause = {
	enable = {
		has_country_flag = FAI_op_pause_timer
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		ordertype = front
		ratio = 0.5
		priority = 2000
		execution_type = careful
		execute_order = no
		manual_attack = no
	}
}

############################################
# Defensive Hold vs Stronger Aggressor (#15)
############################################
FAI_hold_front_recent_wardec = {
	enable = {
		any_enemy_country = {
			PREV = {
				has_country_flag = { flag = FAI_hold_against_@PREV value = 1 days < 60 }
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		ordertype = front
		ratio = 0.5
		priority = 5000
		execution_type = careful
		execute_order = no
		manual_attack = no
		country_trigger = {
			ROOT = {
				has_country_flag = { flag = FAI_hold_against_@FROM value = 1 days < 60 }
			}
		}
	}
}
