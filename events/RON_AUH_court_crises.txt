add_namespace = auh_court

# A recurring pool of stand-alone crisis events for Austria-Hungary, 1900-1909.
# Fired at random by a monthly chance roll wired into RON_AUH_dnc_on_actions.txt.
# Purely additive: these touch only political_power, stability, war_support,
# manpower, army/navy experience, AUH_strike_pressure, and small (1-4 point)
# nudges to the register's nationality support variables. The register itself,
# the Ausgleich engine, the focus tree, the companies and the cast are untouched.

country_event = {
	id = auh_court.1
	title = auh_court.1.t
	desc = auh_court.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	fire_only_once = yes
	trigger = { tag = AUH }
	option = {
		name = auh_court.1.a
		ai_chance = { factor = 35 }
		add_political_power = 30
		add_stability = -0.04
		add_war_support = -0.02
		add_to_variable = { var = AUS_support value = -2 }
		clamp_variable = { var = AUS_support min = 0 max = 100 }
		custom_effect_tooltip = AUH_court_skoda_coverup_tt
	}
	option = {
		name = auh_court.1.b
		ai_chance = { factor = 35 }
		add_political_power = -40
		add_stability = 0.04
		add_army_experience = 10

		clamp_variable = { var = AUS_support min = 0 max = 100 }
		custom_effect_tooltip = AUH_court_skoda_inquiry_tt
	}
	option = {

}
