	id = QIN.1
	title = "The Future of the Boxers?"
	desc = QIN.1.d

		name = "Formalize the Boxer Militias"
		ai_will_do = {
			base = 5
		}
		add_political_power = 50
		add_ideas = QIN_boxer_forces
	}

		name = "Disband the Boxer Militias"
		ai_will_do = {
			base = 5
		}
		delete_unit_template_and_units = {
			division_template = "Boxer Militia"
			disband = yes
		}
	}
}
