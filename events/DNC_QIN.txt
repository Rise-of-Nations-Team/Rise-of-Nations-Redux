# THE FUTURE OF THE BOXERS?
country_event = {
	id = dnc_qing.1
	title = "The Future of the Boxers?"
	desc = dnc_qing.1.d
	
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = "Disband the Boxer Militias"
		ai_chance = {
			base = 5
		}
		delete_unit_template_and_units = {
			division_template = "Boxer Militia"
			disband = yes
		}
		add_political_power = 50
	}
	option = {
		name = "Formalize the Boxer Militias"
		ai_chance = {
			base = 5
		}
		custom_effect_tooltip = QIN_combine_boxer_ideas_tt
		hidden_effect = {
			add_ideas = QIN_formal_boxer_army
			remove_ideas = QIN_spirit_soldiers1
		}
	}
}
