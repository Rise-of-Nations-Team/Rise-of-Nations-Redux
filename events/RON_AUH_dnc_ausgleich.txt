add_namespace = auh_ausgleich
country_event = {
	id = auh_ausgleich.1
	title = auh_ausgleich.1.t
	desc = auh_ausgleich.1.d
	picture = GFX_report_event_hungary_parliament
	is_triggered_only = yes
	option = {
		name = auh_ausgleich.1.a
		ai_chance = { factor = 55 }
		trigger = { has_political_power > 150 }
		custom_effect_tooltip = auh_ausgleich_bill_tt
		hidden_effect = {
			set_temp_variable = { AUH_bill = AUH_renewal_cost }
			multiply_temp_variable = { AUH_bill = -1 }
			add_political_power = AUH_bill
			set_variable = { AUH_next_term = 120 }
			set_variable = { AUH_bill_carried = 0 }
			random = {
				chance = AUH_renewal_odds
				set_variable = { AUH_bill_carried = 1 }
			}
			if = {
				limit = {
					check_variable = {
						var = AUH_bill_carried
						value = 1
						compare = equals
					}
				}
				country_event = { id = auh_ausgleich.2 hours = 4 }
			}
			else = {
				country_event = { id = auh_ausgleich.3 hours = 4 }
			}
		}
	}

	option = {
		name = auh_ausgleich.1.b
		ai_chance = { factor = 25 }
		trigger = { has_political_power > 200 }
		add_political_power = -200
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = AUH_quote_frozen
						value = 1
						compare = equals
					}
				}
			}
			subtract_from_variable = { AUH_quote = 1 }
		}
		add_to_variable = { HUN_support = 8 }
		subtract_from_variable = { AUS_support = 4 }
		subtract_from_variable = { CZE_support = 3 }
		custom_effect_tooltip = auh_ausgleich_bought_tt
		hidden_effect = {
			set_variable = { AUH_next_term = 120 }
			AUH_dnc_ausgleich_clamp = yes
			country_event = { id = auh_ausgleich.2 hours = 4 }
		}
	}

	option = {
		name = auh_ausgleich.1.c
		ai_chance = { factor = 12 }
		add_stability = -0.10
		add_war_support = 0.05
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = AUH_quote_frozen
						value = 1
						compare = equals
					}
				}
			}
			add_to_variable = { AUH_quote = 2 }
		}
		subtract_from_variable = { AUH_gemeinsamkeit = 6 }
		subtract_from_variable = { HUN_support = 12 }
		subtract_from_variable = { CRO_support = 5 }
		add_to_variable = { AUS_support = 3 }
		custom_effect_tooltip = auh_ausgleich_ordinance_tt
		set_country_flag = AUH_dnc_settlement_by_ordinance
		hidden_effect = {
			set_variable = { AUH_next_term = 60 }
			AUH_dnc_the_settlement_carried = yes
		}
	}

	option = {
		name = auh_ausgleich.1.e
		ai_chance = { factor = 8 }
		custom_effect_tooltip = auh_ausgleich_run_out_tt
		hidden_effect = {
			country_event = { id = auh_ausgleich.3 hours = 4 }
		}
	}
}
