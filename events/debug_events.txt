# SETUP EVENTS - hidden events meant for setting things up behind the scenes
country_event = {
	id = setup.1
	title = setup.1.t
	desc = setup.1.d

	is_triggered_only = yes

	option = {
		setup.1.a
		trigger = {
			check_variable = { QIN_box_war_chosen_revolution = 1 }
		}
		set_global_flag = QIN_mongol_revolution
		QIN_box_war_revolutionaries_setup_se = yes
	}
	option = {
		setup.1.b
		trigger = {
			check_variable = { QIN_box_war_chosen_revolution = 2 }
		}
		set_global_flag = QIN_tibetan_revolution
		QIN_box_war_revolutionaries_setup_se = yes
	}
	option = {
		setup.1.c
		trigger = {
			check_variable = { QIN_box_war_chosen_revolution = 3 }
		}
		set_global_flag = QIN_guangxi_revolution
		QIN_box_war_revolutionaries_setup_se = yes
	}
	option = {
		setup.1.f
		trigger = {
			check_variable = { QIN_box_war_chosen_revolution = 4 }
		}
		set_global_flag = QIN_han_revolution
		QIN_box_war_revolutionaries_setup_se = yes
	}
	option = {
		setup.1.e
		trigger = {
			AND = {
				has_global_flag = QIN_mongol_revolution
				has_global_flag = QIN_tibetan_revolution
				has_global_flag = QIN_guangxi_revolution
				has_global_flag = QIN_han_revolution
				NOT = {
					has_global_flag = QIN_manchu_revolution
				}
			}
		}
		set_global_flag = QIN_manchu_revolution
		QIN_box_war_revolutionaries_setup_se = yes
	}
	option = {
		setup.1.g
		trigger = {
			check_variable = { QIN_box_war_chosen_revolution = 5 }
			set_global_flag = QIN_tuvan_revolution
			QIN_box_war_revolutionaries_setup_se = yes
		}
	}
}
