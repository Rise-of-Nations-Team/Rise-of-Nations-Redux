add_namespace = dnc_boxers

# Supreme Boxer Leader Zhang Decheng Dies
country_event = {
	id = dnc_boxers.1
	title = dnc_boxers.1.title
	desc = dnc_boxers.1.desc

	fire_only_once = yes

	trigger = {
		tag = BOX
		date > 1900.7.14
		has_country_leader = {
			character = QIN_zhang_decheng
			ruling_only = yes
		}
	}

	immediate = {
		hidden_effect = {
			retire_character = QIN_zhang_decheng
			promote_character = QIN_cao_futian
		}
	}

	option = {
		name = dnc_boxers.1.a
		retire_character = QIN_zhang_decheng
		promote_character = QIN_cao_futian
	}
}

# Boxer Rebellion Takes Over Beijing!
country_event = {
	id = dnc_boxers.3
	title = dnc_boxers.3.title
	desc = dnc_boxers.3.desc

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = QIN
		date > 1900.6.20
		NOT = {
			has_completed_focus = DNC_QIN_antiwestern_policy
		}
	}

	immediate = {
		hidden_effect = {
			608 = {
				add_dynamic_modifier = { modifier = QIN_boxer_rebellion }
			}
			QIN_boxer_setup = yes
			add_stability = -0.15
		}
	}

	option = {
		name = “Something needs to be done about this.”
		add_stability = -0.15
	}
}

# RAINS END DROUGHT
country_event = {
	id = dnc_boxers.4
	title = "Summer Rains End Drought"
	desc = dnc_boxers.4.desc

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		QIN = {
			AND = {
				has_completed_focus = QIN_DNC_anti_western_policy
				OR = {
					has_war_with = GER
					has_war_with = RUS
					has_war_with = AUH
					has_war_with = GER
					has_war_with = ENG
					has_war_with = FRA
					has_war_with = USA
				}
			}
		}
	}

	option = {
		name = "Oh no..."
		trigger = {
			original_tag = QIN
		}
		delete_unit = {
			division_template = "Boxer Militia"
			id = 556
		}
		custom_effect_tooltip = QIN_boxer_resolve_weakens_tt
	}
	option = {
		name = "The Lord smiles on us today"
		trigger = {
			every_country = {
				limit = {
					has_idea = DNC_eight_nation_alliance
				}
			}
		}
	}
}
